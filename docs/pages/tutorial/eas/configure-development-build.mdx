import React, { useState } from "react";
import { View, Text, TextInput, Button, ScrollView, StyleSheet, Linking, Alert } from "react-native";

export default function App() {
  const [complaint, setComplaint] = useState({ name: "", phone: "", message: "" });

  const announcements = [
    { id: 1, title: "Special Offer!", content: "Flat 20% off on Doodh Soda this week." },
    { id: 2, title: "Store Timing Update", content: "Shop will remain closed on Friday." }
  ];

  const handleSubmit = async () => {
    if (complaint.name && complaint.phone && complaint.message) {
      // 👇 اپنا WhatsApp نمبر یہاں ڈالیں (Pakistan country code = 92)
      const whatsappNumber = "9233215447374";

      const text = `📢 New Complaint from Dark Nebula App\n\n👤 Name: ${complaint.name}\n📞 Phone: ${complaint.phone}\n📝 Message: ${complaint.message}`;

      // wa.me URL
      const url = `https://wa.me/${whatsappNumber}?text=${encodeURIComponent(text)}`;
      // fallback whatsapp:// URL
      const fallbackUrl = `whatsapp://send?phone=${whatsappNumber}&text=${encodeURIComponent(text)}`;

      const supported = await Linking.canOpenURL(url);
      if (supported) {
        Linking.openURL(url);
      } else {
        const fallbackSupported = await Linking.canOpenURL(fallbackUrl);
        if (fallbackSupported) {
          Linking.openURL(fallbackUrl);
        } else {
          Alert.alert("Error", "WhatsApp is not installed on this device.");
        }
      }
    } else {
      Alert.alert("Error", "Please fill all fields before submitting.");
    }
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.header}>Dark Nebula Mart</Text>

      {/* Announcements */}
      <Text style={styles.sectionTitle}>📢 Announcements</Text>
      {announcements.map((a) => (
        <View key={a.id} style={styles.card}>
          <Text style={styles.cardTitle}>{a.title}</Text>
          <Text>{a.content}</Text>
        </View>
      ))}

      {/* Complaints */}
      <Text style={styles.sectionTitle}>📝 Submit Complaint</Text>
      <View>
        <TextInput
          placeholder="Your Name"
          value={complaint.name}
          onChangeText={(text) => setComplaint({ ...complaint, name: text })}
          style={styles.input}
        />
        <TextInput
          placeholder="Phone Number"
          value={complaint.phone}
          onChangeText={(text) => setComplaint({ ...complaint, phone: text })}
          style={styles.input}
        />
        <TextInput
          placeholder="Write your complaint here..."
          value={complaint.message}
          onChangeText={(text) => setComplaint({ ...complaint, message: text })}
          style={[styles.input, { height: 100 }]}
          multiline
        />
        <Button title="Send via WhatsApp" onPress={handleSubmit} />
      </View>

      {/* About Us */}
      <Text style={styles.sectionTitle}>ℹ️ About Us</Text>
      <View style={styles.card}>
        <Text>Shop: Dark Nebula Mart</Text>
        <Text>Location: Near Main Bazaar, Bagh Mahalla, Jhelum</Text>
        <Text>Contact: 033215447374</Text>
        <Text>Email: darknebulamart@gmail.com</Text>
      </View>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, backgroundColor: "#fff" },
  header: { fontSize: 24, fontWeight: "bold", textAlign: "center", marginBottom: 20 },
  sectionTitle: { fontSize: 18, fontWeight: "600", marginVertical: 10 },
  card: { backgroundColor: "#f8f9fa", padding: 10, borderRadius: 8, marginBottom: 10 },
  cardTitle: { fontWeight: "bold", fontSize: 16 },
  input: { borderWidth: 1, borderColor: "#ccc", borderRadius: 5, padding: 8, marginBottom: 10 }
});
